import React, { useEffect, useState, useRef } from "react";
import { useSelector, useDispatch } from "react-redux";
import { getAllMessage, createNewMessage } from "../../../api/message";
import { clearunreadMessagecount } from "../../../api/chat";
import toast from "react-hot-toast";
import moment from "moment";
import styles from './../farmStyles/chatarea.module.scss';

const ChatArea = ({ socket }) => {
  const dispatch = useDispatch();
  const selectedChat = useSelector((state) => state.user.selectedChats);
  const user = useSelector((state) => state.user.profile);

  const [messages, setMessages] = useState([]);
  const [input, setInput] = useState("");
  const messagesEndRef = useRef(null);

  const selectedUser = selectedChat?.members?.find((m) => m._id !== user._id);

  // Fetch messages for selected chat
  const fetchMessages = async () => {
    if (!selectedChat?._id) return;
    try {
      const response = await getAllMessage(selectedChat._id);
      if (response.success) {
        setMessages(response.data);
        // Clear unread messages on fetch
        await clearunreadMessagecount(selectedChat._id);
      } else {
        toast.error("Failed to load messages");
      }
    } catch (error) {
      toast.error(error.message || "Error fetching messages");
    }
  };

  // Send message
  const sendMessage = async (text, image = null) => {
    if (!text.trim() && !image) return;

    const newMsg = {
      chatId: selectedChat._id,
      sender: user._id,
      text,
      image,
    };

    socket.emit("send-message", {
      ...newMsg,
      members: selectedChat.members.map((m) => m._id),
      read: false,
      createdAt: new Date().toISOString(),
    });

    try {
      const response = await createNewMessage(newMsg);
      if (response.success) {
        setInput("");
        fetchMessages();
      } else {
        toast.error("Failed to send message");
      }
    } catch (error) {
      toast.error(error.message || "Error sending message");
    }
  };

  // Listen for incoming messages via socket
  useEffect(() => {
    socket.on("receive-message", (message) => {
      if (selectedChat && message.chatId === selectedChat._id) {
        setMessages((prev) => [...prev, message]);
      }
    });

    return () => {
      socket.off("receive-message");
    };
  }, [selectedChat, socket]);

  // Fetch messages when chat changes
  useEffect(() => {
    fetchMessages();
  }, [selectedChat]);

  // Scroll to bottom on new messages
  useEffect(() => {
    messagesEndRef.current?.scrollIntoView({ behavior: "smooth" });
  }, [messages]);

  // Format timestamp for display
  const formatTime = (time) => {
    return moment(time).format("MMM D, hh:mm A");
  };

  return (
    <div className={styles.chatArea}>
      <div className={styles.chatHeader}>
        Chat with {selectedUser ? `${selectedUser.firstname} ${selectedUser.lastname}` : "User"}
      </div>

      <div className={styles.messages} style={{ overflowY: "auto", maxHeight: "70vh" }}>
        {messages.map((msg) => (
          <div
            key={msg._id || msg.id}
            className={`${styles.message} ${msg.sender === user._id ? styles.me : styles.other}`}
          >
            <div>{msg.text}</div>
            {msg.image && <img src={msg.image} alt="sent media" style={{ maxWidth: "150px" }} />}
            <small>{formatTime(msg.createdAt)}</small>
          </div>
        ))}
        <div ref={messagesEndRef} />
      </div>

      <div className={styles.inputArea}>
        <input
          type="text"
          placeholder="Type a message..."
          value={input}
          onChange={(e) => setInput(e.target.value)}
          onKeyDown={(e) => e.key === "Enter" && sendMessage(input)}
        />
        <button onClick={() => sendMessage(input)}>Send</button>
      </div>
    </div>
  );
};

export default ChatArea;
